CREATE FUNCTION dbo.UserExists (@UserName NVARCHAR(100))
RETURNS BIT
AS
BEGIN
    RETURN CASE WHEN EXISTS (SELECT 1 FROM dbo.Users WHERE UserName = @UserName)
                THEN 1 ELSE 0 END;
END
GO

CREATE FUNCTION dbo.GetUserPasswordHash (@UserName NVARCHAR(100))
RETURNS NVARCHAR(255)
AS
BEGIN
    DECLARE @PasswordHash NVARCHAR(255);
    SELECT @PasswordHash = PasswordHash FROM dbo.Users WHERE UserName = @UserName;
    RETURN @PasswordHash;
END
GO

CREATE FUNCTION dbo.CanManageGroup (@UserID INT, @GroupID INT)
RETURNS BIT
AS
BEGIN
    RETURN CASE
        WHEN @GroupID IS NULL THEN 0
        WHEN EXISTS (SELECT 1 FROM dbo.Groups WHERE GroupID = @GroupID AND AdminID = @UserID) THEN 1
        WHEN EXISTS (SELECT 1 FROM dbo.Users WHERE UserID = @UserID AND PermissionID = 2) THEN 1
        ELSE 0 END;
END
GO

CREATE FUNCTION dbo.IsGroupMember (@UserID INT, @GroupID INT)
RETURNS BIT
AS
BEGIN
    RETURN CASE
        WHEN @GroupID IS NULL THEN 0
        WHEN dbo.CanManageGroup(@UserID, @GroupID) = 1 THEN 1
        WHEN EXISTS (SELECT 1 FROM dbo.GroupMembers WHERE GroupID = @GroupID AND UserID = @UserID) THEN 1
        ELSE 0 END;
END
GO

CREATE FUNCTION dbo.CanModifyTask (@TaskID INT, @ActingUserID INT)
RETURNS BIT
AS
BEGIN
    DECLARE @OwnerID INT;
    DECLARE @GroupID INT;

    SELECT @OwnerID = UserID, @GroupID = GroupID FROM dbo.Tasks WHERE TaskID = @TaskID;

    RETURN CASE
        WHEN @OwnerID IS NULL THEN 0
        WHEN @GroupID IS NULL THEN CASE WHEN @OwnerID = @ActingUserID THEN 1 ELSE 0 END
        ELSE dbo.IsGroupMember(@ActingUserID, @GroupID) END;
END
GO

CREATE FUNCTION dbo.GetSubtasks (@TaskID INT)
RETURNS TABLE
AS
RETURN
(
    SELECT T.TaskID,
           T.TaskParentID,
           T.TaskStatusID,
           T.UserID,
           T.GroupID,
           T.Ranking,
           T.TaskName,
           T.LevelOfImportanceID,
           T.Description,
           T.CreationDate,
           T.Color
    FROM dbo.Tasks AS T
    WHERE T.TaskParentID = @TaskID
);
GO

CREATE PROCEDURE dbo.AddUser
    @UserName     NVARCHAR(100),
    @ShowName     NVARCHAR(100),
    @UserEmail    VARCHAR(255),
    @PasswordHash VARCHAR(255),
    @PermissionID INT,
    @Color        VARCHAR(16) = '#FFFFFF'
AS
BEGIN
    SET XACT_ABORT ON;
    SET NOCOUNT ON;
    BEGIN TRANSACTION;

    IF dbo.UserExists(@UserName) = 1
        THROW 50003, N'This username is already taken.', 1;

    INSERT INTO dbo.Users (UserName, ShowName, UserEmail, PasswordHash, PermissionID, Color)
    VALUES (@UserName, @ShowName,
            CASE WHEN NULLIF(LTRIM(RTRIM(@UserEmail)), N'') IS NULL THEN NULL
                 ELSE LOWER(LTRIM(RTRIM(@UserEmail))) END,
            @PasswordHash, 1, @Color);

    COMMIT TRANSACTION;
    SELECT CAST(SCOPE_IDENTITY() AS INT) AS NewID;
END
GO

CREATE PROCEDURE dbo.UpdatePasswordHash
    @UserID        INT,
    @PasswordHash  VARCHAR(255)
AS
BEGIN
    SET XACT_ABORT ON;
    BEGIN TRANSACTION;

    UPDATE dbo.Users
    SET PasswordHash = @PasswordHash
    WHERE UserID = @UserID;

    IF @@ROWCOUNT = 0
        THROW 50002, N'User was not found.', 1;

    COMMIT TRANSACTION;
END
GO

CREATE PROCEDURE dbo.DeleteUser
    @UserID              INT,
    @ConfirmGroupImpact  BIT = 0
AS
BEGIN
    SET XACT_ABORT ON;
    BEGIN TRANSACTION;

    IF @ConfirmGroupImpact = 0
       AND EXISTS (
           SELECT 1
           FROM dbo.Groups G
           JOIN dbo.GroupMembers M ON M.GroupID = G.GroupID
           WHERE G.AdminID = @UserID
             AND M.UserID <> @UserID)
        THROW 51001, N'Deleting this account also deletes tasks created by other members of the groups you administer. Call again with @ConfirmGroupImpact = 1 to confirm.', 1;

    DECLARE @AdminGroups TABLE (GroupID INT PRIMARY KEY);
    INSERT INTO @AdminGroups (GroupID)
    SELECT GroupID FROM dbo.Groups WHERE AdminID = @UserID;

    DECLARE @TasksToDelete TABLE (TaskID INT PRIMARY KEY);
    INSERT INTO @TasksToDelete (TaskID)
    SELECT TaskID FROM dbo.Tasks
    WHERE UserID = @UserID OR GroupID IN (SELECT GroupID FROM @AdminGroups);

    DECLARE @RowsAdded INT = 1;
    WHILE @RowsAdded > 0
    BEGIN
        INSERT INTO @TasksToDelete (TaskID)
        SELECT T.TaskID FROM dbo.Tasks T
        JOIN @TasksToDelete D ON T.TaskParentID = D.TaskID
        WHERE NOT EXISTS (SELECT 1 FROM @TasksToDelete X WHERE X.TaskID = T.TaskID);

        SET @RowsAdded = @@ROWCOUNT;
    END

    DELETE FROM dbo.Attachments WHERE TaskID IN (SELECT TaskID FROM @TasksToDelete);
    DELETE FROM dbo.Assigned    WHERE TaskID IN (SELECT TaskID FROM @TasksToDelete);
    DELETE FROM dbo.Planned     WHERE TaskID IN (SELECT TaskID FROM @TasksToDelete);
    DELETE FROM dbo.Tasks       WHERE TaskID IN (SELECT TaskID FROM @TasksToDelete);

    DELETE FROM dbo.PostsInGroups WHERE GroupID IN (SELECT GroupID FROM @AdminGroups);
    DELETE FROM dbo.GroupMembers  WHERE GroupID IN (SELECT GroupID FROM @AdminGroups);
    DELETE FROM dbo.Groups        WHERE GroupID IN (SELECT GroupID FROM @AdminGroups);

    DELETE FROM dbo.PostsInGroups WHERE UserID = @UserID;

    DELETE FROM dbo.GroupMembers  WHERE UserID = @UserID;
    DELETE FROM dbo.Assigned      WHERE ToUserID = @UserID;

    DELETE FROM dbo.Users WHERE UserID = @UserID;

    COMMIT TRANSACTION;
END
GO

CREATE PROCEDURE dbo.AddGroup

CREATE PROCEDURE dbo.UpdateGroup
    @GroupID           INT,
    @ActingUserID      INT,
    @GroupName         NVARCHAR(100),
    @GroupDescription  NVARCHAR(MAX) = NULL,
    @Color             VARCHAR(16) = '#FFFFFF'
AS
BEGIN
    SET XACT_ABORT ON;
    BEGIN TRANSACTION;

    IF dbo.CanManageGroup(@ActingUserID, @GroupID) = 0
        THROW 50004, N'Only the group owner can edit this group.', 1;

    UPDATE dbo.Groups
    SET GroupName        = @GroupName,
        GroupDescription = @GroupDescription,
        Color            = @Color
    WHERE GroupID = @GroupID;

    IF @@ROWCOUNT = 0
        THROW 50002, N'Group was not found.', 1;

    COMMIT TRANSACTION;
END
GO

CREATE PROCEDURE dbo.DeleteGroup
    @GroupID      INT,
    @ActingUserID INT
AS
BEGIN
    SET XACT_ABORT ON;
    BEGIN TRANSACTION;

    IF dbo.CanManageGroup(@ActingUserID, @GroupID) = 0
        THROW 50004, N'Only the group owner can delete this group.', 1;

    DECLARE @TasksToDelete TABLE (TaskID INT PRIMARY KEY);
    INSERT INTO @TasksToDelete (TaskID)
    SELECT TaskID FROM dbo.Tasks WHERE GroupID = @GroupID AND TaskParentID IS NULL;

    DECLARE @RowsAdded INT = 1;
    WHILE @RowsAdded > 0
    BEGIN
        INSERT INTO @TasksToDelete (TaskID)
        SELECT T.TaskID FROM dbo.Tasks T
        JOIN @TasksToDelete D ON T.TaskParentID = D.TaskID
        WHERE NOT EXISTS (SELECT 1 FROM @TasksToDelete X WHERE X.TaskID = T.TaskID);

        SET @RowsAdded = @@ROWCOUNT;
    END

    DELETE FROM dbo.Attachments WHERE TaskID IN (SELECT TaskID FROM @TasksToDelete);
    DELETE FROM dbo.Assigned    WHERE TaskID IN (SELECT TaskID FROM @TasksToDelete);
    DELETE FROM dbo.Planned     WHERE TaskID IN (SELECT TaskID FROM @TasksToDelete);
    DELETE FROM dbo.Tasks       WHERE TaskID IN (SELECT TaskID FROM @TasksToDelete);

    DELETE FROM dbo.PostsInGroups WHERE GroupID = @GroupID;
    DELETE FROM dbo.GroupMembers  WHERE GroupID = @GroupID;
    DELETE FROM dbo.Groups        WHERE GroupID = @GroupID;

    COMMIT TRANSACTION;
END
GO

CREATE PROCEDURE dbo.AddGroupMember
    @GroupID      INT,
    @UserID       INT,
    @ActingUserID INT
AS
BEGIN
    SET XACT_ABORT ON;
    BEGIN TRANSACTION;

    IF dbo.CanManageGroup(@ActingUserID, @GroupID) = 0
        THROW 50001, N'You do not have permission to manage this group.', 1;

    IF NOT EXISTS (SELECT 1 FROM dbo.Users WHERE UserID = @UserID)
        THROW 50002, N'User was not found.', 1;

    INSERT INTO dbo.GroupMembers (GroupID, UserID)
    VALUES (@GroupID, @UserID);

    COMMIT TRANSACTION;
END
GO

CREATE PROCEDURE dbo.DeleteGroupMember
    @GroupID      INT,
    @UserID       INT,
    @ActingUserID INT
AS
BEGIN
    SET XACT_ABORT ON;
    BEGIN TRANSACTION;

    IF dbo.CanManageGroup(@ActingUserID, @GroupID) = 0
        AND @ActingUserID <> @UserID
        THROW 50001, N'You do not have permission to manage this group.', 1;

    IF EXISTS (SELECT 1 FROM dbo.Groups WHERE GroupID = @GroupID AND AdminID = @UserID)
        THROW 50004, N'The group owner cannot be removed from the group.', 1;

    DELETE FROM dbo.GroupMembers WHERE GroupID = @GroupID AND UserID = @UserID;

    DELETE FROM dbo.Assigned
    WHERE ToUserID = @UserID
      AND TaskID IN (SELECT TaskID FROM dbo.Tasks WHERE GroupID = @GroupID);

    COMMIT TRANSACTION;
END
GO

CREATE PROCEDURE dbo.AddPostInGroup
    @GroupID       INT,
    @UserID        INT,
    @PostContent   NVARCHAR(MAX)
AS
BEGIN
    SET XACT_ABORT ON;
    SET NOCOUNT ON;
    BEGIN TRANSACTION;

    IF dbo.IsGroupMember(@UserID, @GroupID) = 0
        THROW 50001, N'You must be a member of the group to post.', 1;

    IF NULLIF(LTRIM(RTRIM(@PostContent)), N'') IS NULL
        THROW 50003, N'Post content cannot be empty.', 1;

    INSERT INTO dbo.PostsInGroups (GroupID, UserID, PostContent)
    VALUES (@GroupID, @UserID, @PostContent);

    COMMIT TRANSACTION;
    SELECT CAST(SCOPE_IDENTITY() AS INT) AS NewID;
END
GO

CREATE PROCEDURE dbo.GetPostsInGroup
    @GroupID       INT,
    @ActingUserID  INT,
    @BeforePostID  INT = NULL,
    @Take          INT = 20
AS
BEGIN
    SET NOCOUNT ON;

    IF dbo.IsGroupMember(@ActingUserID, @GroupID) = 0
        THROW 50001, N'You do not have access to this group.', 1;

    SELECT TOP (@Take)
           P.PostID,
           P.GroupID,
           P.UserID,
           P.PostContent,
           P.PostDate,
           P.CreatedAt,
           U.UserName,
           U.ShowName,
           U.Color AS UserColor,
           CAST(CASE WHEN P.UserID = @ActingUserID THEN 1 ELSE 0 END AS BIT) AS IsMine
    FROM dbo.PostsInGroups P
    JOIN dbo.Users U ON P.UserID = U.UserID
    WHERE P.GroupID = @GroupID
      AND (@BeforePostID IS NULL OR P.PostID < @BeforePostID)
    ORDER BY P.PostID DESC;
END
GO

CREATE PROCEDURE dbo.AddTask
    @TaskParentID        INT = NULL,
    @TaskStatusID        INT = 2,
    @UserID              INT,
    @GroupID             INT = NULL,
    @TaskName            NVARCHAR(100),
    @ImportanceLevelID   INT = 2,
    @Description         NVARCHAR(MAX) = NULL,
    @Color               VARCHAR(16) = '#FFFFFF',
    @ActingUserID        INT
AS
BEGIN
    SET XACT_ABORT ON;
    SET NOCOUNT ON;
    BEGIN TRANSACTION;

    IF @UserID <> @ActingUserID
        THROW 50001, N'You can only create tasks for yourself.', 1;

    IF @GroupID IS NOT NULL AND dbo.IsGroupMember(@ActingUserID, @GroupID) = 0
        THROW 50001, N'You must be a member of the group to add tasks.', 1;

    IF @TaskParentID IS NOT NULL
    BEGIN
        DECLARE @ParentOwnerID INT;
        DECLARE @ParentGroupID INT;

        SELECT @ParentOwnerID = UserID, @ParentGroupID = GroupID
        FROM dbo.Tasks WHERE TaskID = @TaskParentID;

        IF @ParentOwnerID IS NULL
            THROW 50002, N'Parent task was not found.', 1;

        IF EXISTS (SELECT 1 FROM dbo.Tasks WHERE TaskID = @TaskParentID AND TaskParentID IS NOT NULL)
            THROW 50003, N'A task cannot have a parent that is itself a step.', 1;

        IF dbo.CanModifyTask(@TaskParentID, @ActingUserID) = 0
            THROW 50001, N'You do not have permission to modify this task.', 1;

        IF ISNULL(@ParentGroupID, -1) <> ISNULL(@GroupID, -1)
            THROW 50003, N'Parent task must belong to the same list.', 1;
    END

    DECLARE @EffectiveStatus INT = @TaskStatusID;

    IF @GroupID IS NOT NULL AND @TaskParentID IS NULL AND dbo.CanManageGroup(@ActingUserID, @GroupID) = 0
        SET @EffectiveStatus = 1;

    DECLARE @NextRanking INT;
    SELECT @NextRanking = ISNULL(MAX(T.Ranking), 0) + 1
    FROM dbo.Tasks T WITH (UPDLOCK, HOLDLOCK)
    WHERE (T.TaskParentID = @TaskParentID OR (T.TaskParentID IS NULL AND @TaskParentID IS NULL))
      AND ((@GroupID IS NULL AND T.UserID = @UserID AND T.GroupID IS NULL)
           OR (@GroupID IS NOT NULL AND T.GroupID = @GroupID));

    INSERT INTO dbo.Tasks
        (TaskParentID, TaskStatusID, UserID, GroupID, Ranking, TaskName, LevelOfImportanceID, Description, Color)
    VALUES
        (@TaskParentID, @EffectiveStatus, @UserID, @GroupID, @NextRanking, @TaskName, @ImportanceLevelID, @Description, @Color);

    COMMIT TRANSACTION;
    SELECT CAST(SCOPE_IDENTITY() AS INT) AS NewID;
END
GO

CREATE PROCEDURE dbo.SaveTask
    @TaskID              INT = NULL,
    @TaskParentID        INT = NULL,
    @TaskStatusID        INT = 2,
    @UserID              INT,
    @GroupID             INT = NULL,
    @TaskName            NVARCHAR(100),
    @ImportanceLevelID   INT = 2,
    @Description         NVARCHAR(MAX) = NULL,
    @Color               VARCHAR(16) = '#FFFFFF',
    @DueStartDate        DATETIME2 = NULL,
    @DueEndDate          DATETIME2 = NULL,
    @RepetitionTypeID    INT = 1,
    @Assignees           dbo.TVPUserIDs READONLY,
    @ActingUserID        INT
AS
BEGIN
    SET XACT_ABORT ON;
    SET NOCOUNT ON;
    BEGIN TRANSACTION;

    IF @DueEndDate < @DueStartDate
        THROW 50003, N'End date cannot be before the start date.', 1;

    IF @GroupID IS NOT NULL AND dbo.IsGroupMember(@ActingUserID, @GroupID) = 0
        THROW 50001, N'You must be a member of the group to modify tasks.', 1;

    IF @GroupID IS NULL AND EXISTS (SELECT 1 FROM @Assignees)
        THROW 50003, N'Personal tasks cannot have assignees.', 1;

    IF @GroupID IS NOT NULL AND EXISTS (
        SELECT 1 FROM @Assignees A WHERE dbo.IsGroupMember(A.UserID, @GroupID) = 0)
        THROW 50003, N'Assignees must be members of the group.', 1;

    IF @TaskID IS NOT NULL AND dbo.CanModifyTask(@TaskID, @ActingUserID) = 0
        THROW 50001, N'You do not have permission to modify this task.', 1;

    DECLARE @IsInsert BIT = CASE WHEN @TaskID IS NULL THEN 1 ELSE 0 END;
    DECLARE @EffectiveStatus INT = @TaskStatusID;

    IF @IsInsert = 1
    BEGIN
        IF @UserID <> @ActingUserID
            THROW 50001, N'You can only create tasks for yourself.', 1;

        IF @TaskParentID IS NOT NULL
        BEGIN
            DECLARE @ParentOwnerID INT;
            DECLARE @ParentGroupID INT;

            SELECT @ParentOwnerID = UserID, @ParentGroupID = GroupID
            FROM dbo.Tasks WHERE TaskID = @TaskParentID;

            IF @ParentOwnerID IS NULL
                THROW 50002, N'Parent task was not found.', 1;

            IF EXISTS (SELECT 1 FROM dbo.Tasks WHERE TaskID = @TaskParentID AND TaskParentID IS NOT NULL)
                THROW 50003, N'A task cannot have a parent that is itself a step.', 1;

            IF dbo.CanModifyTask(@TaskParentID, @ActingUserID) = 0
                THROW 50001, N'You do not have permission to modify this task.', 1;

            IF ISNULL(@ParentGroupID, -1) <> ISNULL(@GroupID, -1)
                THROW 50003, N'Parent task must belong to the same list.', 1;
        END

        IF @GroupID IS NOT NULL AND @TaskParentID IS NULL AND dbo.CanManageGroup(@ActingUserID, @GroupID) = 0
            SET @EffectiveStatus = 1;

        DECLARE @NextRanking INT;
        SELECT @NextRanking = ISNULL(MAX(T.Ranking), 0) + 1
        FROM dbo.Tasks T WITH (UPDLOCK, HOLDLOCK)
        WHERE (T.TaskParentID = @TaskParentID OR (T.TaskParentID IS NULL AND @TaskParentID IS NULL))
          AND ((@GroupID IS NULL AND T.UserID = @UserID AND T.GroupID IS NULL)
               OR (@GroupID IS NOT NULL AND T.GroupID = @GroupID));

        INSERT INTO dbo.Tasks
            (TaskParentID, TaskStatusID, UserID, GroupID, Ranking, TaskName, LevelOfImportanceID, Description, Color)
        VALUES
            (@TaskParentID, @EffectiveStatus, @UserID, @GroupID, @NextRanking, @TaskName, @ImportanceLevelID, @Description, @Color);

        SET @TaskID = SCOPE_IDENTITY();
    END
    ELSE
    BEGIN
        DECLARE @CurrentUserID INT;
        DECLARE @CurrentGroupID INT;
        DECLARE @CurrentParentID INT;

        SELECT @CurrentUserID = UserID, @CurrentGroupID = GroupID, @CurrentParentID = TaskParentID
        FROM dbo.Tasks WHERE TaskID = @TaskID;

        IF @CurrentUserID IS NULL
            THROW 50002, N'Task was not found.', 1;

        IF ISNULL(@UserID, -1) <> ISNULL(@CurrentUserID, -1)
            OR ISNULL(@GroupID, -1) <> ISNULL(@CurrentGroupID, -1)
            OR ISNULL(@TaskParentID, -1) <> ISNULL(@CurrentParentID, -1)
            THROW 50003, N'Task owner, group or parent must match the stored task.', 1;

        UPDATE dbo.Tasks
        SET TaskName            = @TaskName,
            LevelOfImportanceID = @ImportanceLevelID,
            Description         = @Description,
            Color               = @Color
        WHERE TaskID = @TaskID;
    END

    IF @DueStartDate IS NOT NULL
    BEGIN
        IF EXISTS (SELECT 1 FROM dbo.Planned WHERE TaskID = @TaskID)
            UPDATE dbo.Planned
            SET PlannedStartDate = @DueStartDate,
                PlannedEndDate   = @DueEndDate,
                RepetitionTypeID = @RepetitionTypeID
            WHERE TaskID = @TaskID;
        ELSE
            INSERT INTO dbo.Planned (TaskID, PlannedStartDate, PlannedEndDate, RepetitionTypeID)
            VALUES (@TaskID, @DueStartDate, @DueEndDate, @RepetitionTypeID);
    END
    ELSE
        DELETE FROM dbo.Planned WHERE TaskID = @TaskID;

    DELETE FROM dbo.Assigned WHERE TaskID = @TaskID;
    INSERT INTO dbo.Assigned (TaskID, ToUserID)
    SELECT @TaskID, A.UserID FROM @Assignees A;

    COMMIT TRANSACTION;

    SELECT @TaskID AS NewID;
END
GO

CREATE PROCEDURE dbo.UpdateTaskStatus
    @TaskID        INT,
    @TaskStatusID  INT,
    @ActingUserID  INT
AS
BEGIN
    SET XACT_ABORT ON;
    SET NOCOUNT ON;
    BEGIN TRANSACTION;

    IF dbo.CanModifyTask(@TaskID, @ActingUserID) = 0
        THROW 50001, N'You do not have permission to modify this task.', 1;

    DECLARE @GroupID INT;
    DECLARE @CurrentStatus INT;
    SELECT @CurrentStatus = TaskStatusID, @GroupID = GroupID
    FROM dbo.Tasks WHERE TaskID = @TaskID;

    IF @CurrentStatus IS NULL
        THROW 50002, N'Task was not found.', 1;

    IF @CurrentStatus = 1 AND @TaskStatusID = 3
        THROW 50004, N'A pending task cannot be completed before it is approved.', 1;

    DECLARE @RequestedStatus INT = @TaskStatusID;

    IF @CurrentStatus = 1 AND dbo.CanManageGroup(@ActingUserID, @GroupID) = 0
        SET @TaskStatusID = 1;

    DECLARE @StatusForced BIT =
        CASE WHEN @TaskStatusID <> @RequestedStatus THEN 1 ELSE 0 END;

    DECLARE @OldStart DATETIME2;
    DECLARE @OldEnd   DATETIME2;
    DECLARE @RepetitionTypeID INT;

    SELECT @OldStart = P.PlannedStartDate, @OldEnd = P.PlannedEndDate, @RepetitionTypeID = P.RepetitionTypeID
    FROM dbo.Planned P
    WHERE P.TaskID = @TaskID;

    IF @TaskStatusID = 3 AND @RepetitionTypeID IS NOT NULL AND @RepetitionTypeID > 1 AND @OldStart IS NOT NULL
    BEGIN
        -- EDGE CASE: DATEADD(MONTH/YEAR, ...) silently clamps Jan 31 to Feb 28 and Feb 29 to
        -- Feb 28, which would drag the recurrence to the 28th forever. When the scheduled day is
        -- the last day of its month, every candidate is re-anchored on the last day of the month
        -- it lands in, so the original day (Jan 31, Feb 29) is kept across short and leap years.
        DECLARE @IsMonthEnd BIT =
            CASE WHEN @RepetitionTypeID >= 4
                      AND DATEPART(DAY, @OldStart) = DATEPART(DAY, EOMONTH(@OldStart))
                 THEN 1 ELSE 0 END;

        DECLARE @n INT;
        SET @n = CASE @RepetitionTypeID
            WHEN 2 THEN DATEDIFF(DAY,   CAST(@OldStart AS DATE), CAST(GETDATE() AS DATE))
            WHEN 3 THEN DATEDIFF(DAY,   CAST(@OldStart AS DATE), CAST(GETDATE() AS DATE)) / 7
            WHEN 4 THEN DATEDIFF(MONTH, @OldStart, GETDATE())
            WHEN 5 THEN DATEDIFF(YEAR,  @OldStart, GETDATE())
            ELSE 0 END;
        IF @n < 0
            SET @n = 0;

        DECLARE @NewStart DATETIME2;
        IF @RepetitionTypeID BETWEEN 2 AND 5
        BEGIN
            WHILE @n >= 0
            BEGIN
                SET @NewStart = CASE @RepetitionTypeID
                    WHEN 2 THEN DATEADD(DAY,   @n, @OldStart)
                    WHEN 3 THEN DATEADD(WEEK,  @n, @OldStart)
                    WHEN 4 THEN DATEADD(MONTH, @n, @OldStart)
                    WHEN 5 THEN DATEADD(YEAR,  @n, @OldStart)
                    ELSE @OldStart END;

                IF @IsMonthEnd = 1
                    SET @NewStart = DATEADD(DAY, DATEDIFF(DAY, CAST(@NewStart AS DATE), EOMONTH(@NewStart)), @NewStart);

                -- Only a start that is already in the past is rolled forward, so an occurrence
                -- that is still ahead (early completion) is kept, while one that is due exactly
                -- now advances to the next period exactly as before.
                IF @NewStart > GETDATE()
                    BREAK;

                SET @n = @n + 1;
            END
        END

        IF @NewStart IS NULL
            SET @NewStart = @OldStart;

        SET @OldEnd = CASE WHEN @OldEnd IS NULL THEN NULL
                           ELSE DATEADD(DAY, DATEDIFF(DAY, @OldStart, @NewStart), @OldEnd) END;

        UPDATE dbo.Planned
        SET PlannedStartDate = @NewStart,
            PlannedEndDate   = @OldEnd
        WHERE TaskID = @TaskID;

        DECLARE @NextOccurrenceStatus INT = 2;
        UPDATE dbo.Tasks
        SET TaskStatusID = @NextOccurrenceStatus
        WHERE TaskID = @TaskID;

        SELECT @NextOccurrenceStatus AS TaskStatusID,
               @NewStart              AS PlannedStartDate,
               @OldEnd                AS PlannedEndDate,
               @StatusForced          AS StatusForced;

        COMMIT TRANSACTION;
        RETURN;
    END

    UPDATE dbo.Tasks
    SET TaskStatusID = @TaskStatusID
    WHERE TaskID = @TaskID;

    SELECT @TaskStatusID AS TaskStatusID,
           @OldStart     AS PlannedStartDate,
           @OldEnd       AS PlannedEndDate,
           @StatusForced AS StatusForced;

    COMMIT TRANSACTION;
END
GO

CREATE PROCEDURE dbo.DeleteTask
    @TaskID       INT,
    @ActingUserID INT
AS
BEGIN
    SET XACT_ABORT ON;
    BEGIN TRANSACTION;

    IF NOT EXISTS (
        SELECT 1 FROM dbo.Tasks T
        WHERE T.TaskID = @TaskID
          AND (T.UserID = @ActingUserID OR dbo.CanManageGroup(@ActingUserID, T.GroupID) = 1))
        THROW 50001, N'You do not have permission to delete this task.', 1;

    DECLARE @TasksToDelete TABLE (TaskID INT PRIMARY KEY);
    INSERT INTO @TasksToDelete (TaskID) VALUES (@TaskID);

    DECLARE @RowsAdded INT = 1;
    WHILE @RowsAdded > 0
    BEGIN
        INSERT INTO @TasksToDelete (TaskID)
        SELECT T.TaskID FROM dbo.Tasks T
        JOIN @TasksToDelete D ON T.TaskParentID = D.TaskID
        WHERE NOT EXISTS (SELECT 1 FROM @TasksToDelete X WHERE X.TaskID = T.TaskID);

        SET @RowsAdded = @@ROWCOUNT;
    END

    DELETE FROM dbo.Attachments WHERE TaskID IN (SELECT TaskID FROM @TasksToDelete);
    DELETE FROM dbo.Assigned    WHERE TaskID IN (SELECT TaskID FROM @TasksToDelete);
    DELETE FROM dbo.Planned     WHERE TaskID IN (SELECT TaskID FROM @TasksToDelete);
    DELETE FROM dbo.Tasks       WHERE TaskID IN (SELECT TaskID FROM @TasksToDelete);

    COMMIT TRANSACTION;
END
GO

CREATE PROCEDURE dbo.ReorderTasks
    @UserID       INT,
    @GroupID      INT = NULL,
    @TaskIDs      dbo.TVPTaskIDs READONLY
AS
BEGIN
    SET XACT_ABORT ON;
    SET NOCOUNT ON;
    BEGIN TRANSACTION;

    IF NOT (@GroupID IS NULL OR dbo.IsGroupMember(@UserID, @GroupID) = 1)
        THROW 50001, N'You do not have permission to reorder this list.', 1;

    IF EXISTS (
        SELECT 1
        FROM @TaskIDs I
        LEFT JOIN dbo.Tasks T WITH (UPDLOCK, HOLDLOCK) ON T.TaskID = I.TaskID
        WHERE T.TaskID IS NULL
           OR T.TaskParentID IS NOT NULL
           OR NOT ((@GroupID IS NULL AND T.UserID = @UserID AND T.GroupID IS NULL)
                   OR (@GroupID IS NOT NULL AND T.GroupID = @GroupID)))
        THROW 50002, N'One or more tasks do not belong to this list.', 1;

    ;WITH ScopeTasks AS
    (
        SELECT T.TaskID,
               T.Ranking,
               I.SortOrder,
               CASE WHEN I.TaskID IS NULL THEN 1 ELSE 0 END AS IsUnlisted
        FROM dbo.Tasks T WITH (UPDLOCK, HOLDLOCK)
        LEFT JOIN @TaskIDs I ON I.TaskID = T.TaskID
        WHERE T.TaskParentID IS NULL
          AND ((@GroupID IS NULL AND T.UserID = @UserID AND T.GroupID IS NULL)
               OR (@GroupID IS NOT NULL AND T.GroupID = @GroupID))
    ),
    Ordered AS
    (
        SELECT TaskID,
               ROW_NUMBER() OVER (ORDER BY IsUnlisted, SortOrder, Ranking) AS NewRanking
        FROM ScopeTasks
    )
    UPDATE T
    SET T.Ranking = O.NewRanking
    FROM dbo.Tasks T
    JOIN Ordered O ON T.TaskID = O.TaskID;

    COMMIT TRANSACTION;
END
GO

CREATE PROCEDURE dbo.GetTaskCounts
    @UserID           INT,
    @IncludeCompleted BIT = 0
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @Today DATE = CAST(GETDATE() AS DATE);
    DECLARE @WeekStart DATE = DATEADD(DAY, -((DATEPART(WEEKDAY, @Today) + @@DATEFIRST - 2) % 7), @Today);
    DECLARE @WeekEnd DATE = DATEADD(DAY, 6, @WeekStart);
    DECLARE @MonthStart DATE = DATEADD(DAY, 1 - DAY(GETDATE()), CAST(GETDATE() AS DATE));
    DECLARE @MonthEnd DATE = EOMONTH(GETDATE());
    DECLARE @YearStart DATE = DATEFROMPARTS(YEAR(GETDATE()), 1, 1);
    DECLARE @YearEnd DATE = DATEFROMPARTS(YEAR(GETDATE()), 12, 31);

    -- @IncludeCompleted mirrors the "Show completed" checkbox, so the sidebar
    -- badges always describe exactly the set of tasks rendered in the list.
    DECLARE @AllCount INT = (
        SELECT COUNT(*) FROM dbo.Tasks
        WHERE TaskParentID IS NULL
          AND (@IncludeCompleted = 1 OR TaskStatusID <> 3)
          AND (UserID = @UserID AND GroupID IS NULL));

    DECLARE @ImportantCount INT = (
        SELECT COUNT(*) FROM dbo.Tasks
        WHERE TaskParentID IS NULL AND LevelOfImportanceID = 3
          AND (@IncludeCompleted = 1 OR TaskStatusID <> 3)
          AND (UserID = @UserID AND GroupID IS NULL));

    DECLARE @TodayCount INT = (
        SELECT COUNT(*) FROM dbo.Tasks T
        WHERE T.TaskParentID IS NULL
          AND (@IncludeCompleted = 1 OR T.TaskStatusID <> 3)
          AND (T.UserID = @UserID AND T.GroupID IS NULL)
          AND EXISTS (SELECT 1 FROM dbo.Planned P
                      WHERE P.TaskID = T.TaskID
                        AND CAST(P.PlannedStartDate AS DATE) <= @Today
                        AND (P.PlannedEndDate IS NULL OR CAST(P.PlannedEndDate AS DATE) >= @Today)));

    DECLARE @WeeklyCount INT = (
        SELECT COUNT(*) FROM dbo.Tasks T
        WHERE T.TaskParentID IS NULL
          AND (@IncludeCompleted = 1 OR T.TaskStatusID <> 3)
          AND (T.UserID = @UserID AND T.GroupID IS NULL)
          AND EXISTS (SELECT 1 FROM dbo.Planned P
                      WHERE P.TaskID = T.TaskID
                        AND CAST(P.PlannedStartDate AS DATE) <= @WeekEnd
                        AND COALESCE(CAST(P.PlannedEndDate AS DATE), CAST(P.PlannedStartDate AS DATE)) >= @WeekStart));

    DECLARE @MonthlyCount INT = (
        SELECT COUNT(*) FROM dbo.Tasks T
        WHERE T.TaskParentID IS NULL
          AND (@IncludeCompleted = 1 OR T.TaskStatusID <> 3)
          AND (T.UserID = @UserID AND T.GroupID IS NULL)
          AND EXISTS (SELECT 1 FROM dbo.Planned P
                      WHERE P.TaskID = T.TaskID
                        AND CAST(P.PlannedStartDate AS DATE) <= @MonthEnd
                        AND COALESCE(CAST(P.PlannedEndDate AS DATE), CAST(P.PlannedStartDate AS DATE)) >= @MonthStart));

    DECLARE @YearlyCount INT = (
        SELECT COUNT(*) FROM dbo.Tasks T
        WHERE T.TaskParentID IS NULL
          AND (@IncludeCompleted = 1 OR T.TaskStatusID <> 3)
          AND (T.UserID = @UserID AND T.GroupID IS NULL)
          AND EXISTS (SELECT 1 FROM dbo.Planned P
                      WHERE P.TaskID = T.TaskID
                        AND CAST(P.PlannedStartDate AS DATE) <= @YearEnd
                        AND COALESCE(CAST(P.PlannedEndDate AS DATE), CAST(P.PlannedStartDate AS DATE)) >= @YearStart));

    DECLARE @PlannedCount INT = (
        SELECT COUNT(*) FROM dbo.Tasks T
        WHERE T.TaskParentID IS NULL
          AND (@IncludeCompleted = 1 OR T.TaskStatusID <> 3)
          AND (T.UserID = @UserID AND T.GroupID IS NULL)
          AND EXISTS (SELECT 1 FROM dbo.Planned P WHERE P.TaskID = T.TaskID));

    DECLARE @AssignedCount INT = (
        SELECT COUNT(*) FROM dbo.Tasks T
        JOIN dbo.Assigned A ON T.TaskID = A.TaskID
        WHERE T.TaskParentID IS NULL
          AND (@IncludeCompleted = 1 OR T.TaskStatusID <> 3)
          AND A.ToUserID = @UserID);

    SELECT @AllCount       AS AllCount,
           @TodayCount     AS TodayCount,
           @ImportantCount AS ImportantCount,
           @PlannedCount   AS PlannedCount,
           @AssignedCount  AS AssignedCount,
           @WeeklyCount    AS WeeklyCount,
           @MonthlyCount   AS MonthlyCount,
           @YearlyCount    AS YearlyCount;
END
GO

-- CHANGED (AutoUpdatePlanneds): monthly/yearly rolls re-anchor last-day-of-month recurrences
-- (Jan 31, Feb 29) on the last day of the month they land in instead of drifting to the 28th.
CREATE PROCEDURE dbo.AutoUpdatePlanneds
    @UserID INT = NULL
AS
BEGIN
    SET XACT_ABORT ON;
    SET NOCOUNT ON;

    DECLARE @Rolled TABLE
    (
        TaskID   INT PRIMARY KEY,
        NewStart DATETIME2 NOT NULL,
        NewEnd   DATETIME2 NULL
    );

    INSERT INTO @Rolled (TaskID, NewStart, NewEnd)
    SELECT P.TaskID,
           MonthEnd.NewStart,
           CASE WHEN P.PlannedEndDate IS NULL THEN NULL
                ELSE DATEADD(DAY, DATEDIFF(DAY, P.PlannedStartDate, MonthEnd.NewStart), P.PlannedEndDate)
           END
    FROM dbo.Planned P
    CROSS APPLY
    (
        SELECT n = CASE P.RepetitionTypeID
                    WHEN 2 THEN DATEDIFF(DAY,   CAST(P.PlannedStartDate AS DATE), CAST(GETDATE() AS DATE))
                    WHEN 3 THEN DATEDIFF(DAY,   CAST(P.PlannedStartDate AS DATE), CAST(GETDATE() AS DATE)) / 7
                    WHEN 4 THEN DATEDIFF(MONTH, P.PlannedStartDate, GETDATE())
                    WHEN 5 THEN DATEDIFF(YEAR,  P.PlannedStartDate, GETDATE())
                    ELSE 0 END
    ) N
    CROSS APPLY
    (
        SELECT Candidate = CASE P.RepetitionTypeID
                            WHEN 2 THEN DATEADD(DAY,   N.n, P.PlannedStartDate)
                            WHEN 3 THEN DATEADD(WEEK,  N.n, P.PlannedStartDate)
                            WHEN 4 THEN DATEADD(MONTH, N.n, P.PlannedStartDate)
                            WHEN 5 THEN DATEADD(YEAR,  N.n, P.PlannedStartDate)
                            ELSE P.PlannedStartDate END
    ) C
    CROSS APPLY
    (
        SELECT NewStart = CASE WHEN CAST(C.Candidate AS DATE) < CAST(GETDATE() AS DATE) THEN
                                CASE P.RepetitionTypeID
                                    WHEN 2 THEN DATEADD(DAY,   N.n + 1, P.PlannedStartDate)
                                    WHEN 3 THEN DATEADD(WEEK,  N.n + 1, P.PlannedStartDate)
                                    WHEN 4 THEN DATEADD(MONTH, N.n + 1, P.PlannedStartDate)
                                    WHEN 5 THEN DATEADD(YEAR,  N.n + 1, P.PlannedStartDate)
                                    ELSE P.PlannedStartDate END
                           ELSE C.Candidate END
    ) Nxt
    -- EDGE CASE: DATEADD(MONTH/YEAR, ...) clamps Jan 31 to Feb 28 and Feb 29 to Feb 28, so a
    -- month-end (or leap day) recurrence would stay on the 28th from then on. Re-anchoring those
    -- candidates on the last day of their month keeps Jan 31 / Feb 29 across short and leap years.
    CROSS APPLY
    (
        SELECT NewStart = CASE WHEN P.RepetitionTypeID >= 4
                                 AND DATEPART(DAY, P.PlannedStartDate) = DATEPART(DAY, EOMONTH(P.PlannedStartDate))
                                THEN DATEADD(DAY, DATEDIFF(DAY, CAST(Nxt.NewStart AS DATE), EOMONTH(Nxt.NewStart)), Nxt.NewStart)
                                ELSE Nxt.NewStart END
    ) MonthEnd
    WHERE P.RepetitionTypeID > 1
      AND CAST(P.PlannedStartDate AS DATE) < CAST(GETDATE() AS DATE)
      AND (@UserID IS NULL OR EXISTS (SELECT 1 FROM dbo.Tasks T WHERE T.TaskID = P.TaskID AND T.UserID = @UserID));

    UPDATE P
    SET PlannedStartDate = R.NewStart,
        PlannedEndDate   = R.NewEnd
    FROM dbo.Planned P
    JOIN @Rolled R ON P.TaskID = R.TaskID;

    UPDATE T
    SET T.TaskStatusID = 2
    FROM dbo.Tasks T
    JOIN @Rolled R ON T.TaskID = R.TaskID
    WHERE T.TaskStatusID = 3;
END
GO

CREATE PROCEDURE dbo.GetTasksPlannedByDate
    @UserID INT
AS
BEGIN
    SET NOCOUNT ON;

    SELECT T.* FROM dbo.Tasks T
    JOIN dbo.Planned P ON T.TaskID = P.TaskID
    WHERE T.UserID = @UserID AND T.GroupID IS NULL AND T.TaskParentID IS NULL
    ORDER BY CAST(P.PlannedStartDate AS DATE), T.Ranking;
END
GO

CREATE PROCEDURE dbo.GetTasksToday
    @UserID INT
AS
BEGIN
    SET NOCOUNT ON;

    SELECT T.* FROM dbo.Tasks T
    JOIN dbo.Planned P ON T.TaskID = P.TaskID
    WHERE T.UserID = @UserID AND T.GroupID IS NULL AND T.TaskParentID IS NULL
      AND CAST(P.PlannedStartDate AS DATE) <= CAST(GETDATE() AS DATE)
      AND (P.PlannedEndDate IS NULL OR CAST(P.PlannedEndDate AS DATE) >= CAST(GETDATE() AS DATE))
    ORDER BY CAST(P.PlannedStartDate AS DATE), T.Ranking;
END
GO

CREATE PROCEDURE dbo.GetTasksWeekly
    @UserID INT
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @Today DATE = CAST(GETDATE() AS DATE);
    DECLARE @WeekStart DATE = DATEADD(DAY, -((DATEPART(WEEKDAY, @Today) + @@DATEFIRST - 2) % 7), @Today);
    DECLARE @WeekEnd DATE = DATEADD(DAY, 6, @WeekStart);

    SELECT T.* FROM dbo.Tasks T
    JOIN dbo.Planned P ON T.TaskID = P.TaskID
    WHERE T.UserID = @UserID AND T.GroupID IS NULL AND T.TaskParentID IS NULL
      AND CAST(P.PlannedStartDate AS DATE) <= @WeekEnd
      AND COALESCE(CAST(P.PlannedEndDate AS DATE), CAST(P.PlannedStartDate AS DATE)) >= @WeekStart
    ORDER BY T.Ranking;
END
GO

CREATE PROCEDURE dbo.GetTasksMonthly
    @UserID INT
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @MonthStart DATE = DATEADD(DAY, 1 - DAY(GETDATE()), CAST(GETDATE() AS DATE));
    DECLARE @MonthEnd DATE = EOMONTH(GETDATE());

    SELECT T.* FROM dbo.Tasks T
    JOIN dbo.Planned P ON T.TaskID = P.TaskID
    WHERE T.UserID = @UserID AND T.GroupID IS NULL AND T.TaskParentID IS NULL
      AND CAST(P.PlannedStartDate AS DATE) <= @MonthEnd
      AND COALESCE(CAST(P.PlannedEndDate AS DATE), CAST(P.PlannedStartDate AS DATE)) >= @MonthStart
    ORDER BY T.Ranking;
END
GO

CREATE PROCEDURE dbo.GetTasksYearly
    @UserID INT
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @YearStart DATE = DATEFROMPARTS(YEAR(GETDATE()), 1, 1);

    SELECT T.* FROM dbo.Tasks T

    WHERE T.UserID = @UserID AND T.GroupID IS NULL AND T.TaskParentID IS NULL
      AND CAST(P.PlannedStartDate AS DATE) <= @YearEnd
      AND COALESCE(CAST(P.PlannedEndDate AS DATE), CAST(P.PlannedStartDate AS DATE)) >= @YearStart
    ORDER BY T.Ranking;
END
GO

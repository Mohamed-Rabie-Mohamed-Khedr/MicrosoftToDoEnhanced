CREATE PROCEDURE DeleteUser
    @UserID INT
AS
BEGIN
    BEGIN TRY
        BEGIN TRANSACTION
        DECLARE @AdminGroups TABLE (GroupID INT PRIMARY KEY);
        INSERT INTO @AdminGroups (GroupID)
        SELECT GroupID FROM Groups WHERE AdminID = @UserID;

        DECLARE @TasksToDelete TABLE (TaskID INT PRIMARY KEY);
        INSERT INTO @TasksToDelete (TaskID)
        SELECT TaskID FROM Tasks
        WHERE UserID = @UserID OR GroupID IN (SELECT GroupID FROM @AdminGroups);

        DECLARE @RowsAdded INT = 1;
        WHILE @RowsAdded > 0
        BEGIN
            INSERT INTO @TasksToDelete (TaskID)
            SELECT T.TaskID FROM Tasks T
            JOIN @TasksToDelete D ON T.TaskParentID = D.TaskID
            WHERE T.TaskID NOT IN (SELECT TaskID FROM @TasksToDelete);

            SET @RowsAdded = @@ROWCOUNT;
        END

        DELETE FROM Attachments WHERE TaskID IN (SELECT TaskID FROM @TasksToDelete);
        DELETE FROM Assigned    WHERE TaskID IN (SELECT TaskID FROM @TasksToDelete);
        DELETE FROM Planned     WHERE TaskID IN (SELECT TaskID FROM @TasksToDelete);
        DELETE FROM Tasks       WHERE TaskID IN (SELECT TaskID FROM @TasksToDelete);

        DELETE FROM PostsInGroups WHERE GroupID IN (SELECT GroupID FROM @AdminGroups);
        DELETE FROM GroupMembers  WHERE GroupID IN (SELECT GroupID FROM @AdminGroups);
        DELETE FROM Groups        WHERE GroupID IN (SELECT GroupID FROM @AdminGroups);

        DELETE FROM PostsInGroups WHERE UserID = @UserID;
        DELETE FROM GroupMembers  WHERE UserID = @UserID;
        DELETE FROM Assigned      WHERE ToUserID = @UserID;

        DELETE FROM Users WHERE UserID = @UserID;
        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        ROLLBACK TRANSACTION;
    END CATCH
END
GO

CREATE FUNCTION UserExists (@UserName VARCHAR(100))
RETURNS BIT
AS
BEGIN
    DECLARE @Exists BIT = 0;
    IF EXISTS (SELECT 1 FROM Users WHERE UserName = @UserName)
        SET @Exists = 1;
    RETURN @Exists;
END
GO

CREATE PROCEDURE DeleteGroup
    @GroupID INT
AS
BEGIN
    BEGIN TRY
        BEGIN TRANSACTION
        DECLARE @TasksToDelete TABLE (TaskID INT PRIMARY KEY);
        INSERT INTO @TasksToDelete (TaskID)
        SELECT TaskID FROM Tasks WHERE GroupID = @GroupID AND TaskParentID IS NULL;

        DECLARE @RowsAdded INT = 1;
        WHILE @RowsAdded > 0
        BEGIN
            INSERT INTO @TasksToDelete (TaskID)
            SELECT T.TaskID FROM Tasks T
            JOIN @TasksToDelete D ON T.TaskParentID = D.TaskID
            WHERE T.TaskID NOT IN (SELECT TaskID FROM @TasksToDelete);

            SET @RowsAdded = @@ROWCOUNT;
        END

        DELETE FROM Attachments WHERE TaskID IN (SELECT TaskID FROM @TasksToDelete);
        DELETE FROM Assigned    WHERE TaskID IN (SELECT TaskID FROM @TasksToDelete);
        DELETE FROM Planned     WHERE TaskID IN (SELECT TaskID FROM @TasksToDelete);
        DELETE FROM Tasks       WHERE TaskID IN (SELECT TaskID FROM @TasksToDelete);

        DELETE FROM GroupMembers WHERE GroupID = @GroupID;
        DELETE FROM PostsInGroups WHERE GroupID = @GroupID;
        DELETE FROM Groups WHERE GroupID = @GroupID;

        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        ROLLBACK TRANSACTION;
    END CATCH
END

CREATE PROCEDURE GetPreviousCommentsInGroup
    @GroupID INT,
    @StartCount INT
AS
BEGIN
    DECLARE @PostCount INT = 10;
    DECLARE @EndCount INT = @StartCount - @PostCount;
    IF @EndCount < 1 SET @EndCount = 1;

    SELECT * FROM PostsInGroups
    WHERE GroupID = @GroupID AND PostID BETWEEN @EndCount AND @StartCount
    ORDER BY PostID DESC;
END

CREATE TABLE Tasks (
    TaskID              INT IDENTITY(1,1) PRIMARY KEY,
    TaskParentID        INT NULL,
    TaskStatusID        INT NOT NULL DEFAULT 2,
    UserID              INT NOT NULL,
    GroupID             INT NULL,
    Ranking             INT NOT NULL,
    TaskName            NVARCHAR(100) NOT NULL,
    LevelOfImportanceID INT NOT NULL DEFAULT 2,
    Description         NVARCHAR(MAX) NULL,
    CreationDate        DATE NOT NULL DEFAULT CAST(GETDATE() AS DATE),
    Color               VARCHAR(16) NULL DEFAULT '#FFFFFF',
    FOREIGN KEY (TaskParentID) REFERENCES Tasks(TaskID),
    FOREIGN KEY (TaskStatusID) REFERENCES TaskStatus(TaskStatusID),
    FOREIGN KEY (LevelOfImportanceID) REFERENCES LevelOfImportance(LevelOfImportanceID),
    FOREIGN KEY (UserID) REFERENCES Users(UserID),
    FOREIGN KEY (GroupID) REFERENCES Groups(GroupID)
);
GO

CREATE PROCEDURE AddTask
    @TaskParentID INT = NULL,
    @TaskStatusID INT = 2,
    @UserID INT,
    @GroupID INT = NULL,
    @TaskName NVARCHAR(100),
    @ImportanceLevelID INT = 2,
    @Description NVARCHAR(MAX) = NULL,
    @Color VARCHAR(16) = '#FFFFFF'
AS
BEGIN
    DECLARE @NextRanking INT;
    SELECT @NextRanking = ISNULL(MAX(Ranking), 0) + 1 FROM Tasks WHERE UserID = @UserID;

    INSERT INTO Tasks
        (TaskParentID, TaskStatusID, UserID, GroupID, Ranking, TaskName, LevelOfImportanceID, Description, Color)
    VALUES
        (@TaskParentID, @TaskStatusID, @UserID, @GroupID, @NextRanking, @TaskName, @ImportanceLevelID, @Description, @Color);
END
GO

CREATE PROCEDURE AutoUpdatePlanneds
AS
BEGIN
    UPDATE Planned
    SET PlannedStartDate =
        CASE
            WHEN RepetitionTypeID = 2 THEN DATEADD(DAY, 1, PlannedStartDate)
            WHEN RepetitionTypeID = 3 THEN DATEADD(WEEK, 1, PlannedStartDate)
            WHEN RepetitionTypeID = 4 THEN DATEADD(MONTH, 1, PlannedStartDate)
            WHEN RepetitionTypeID = 5 THEN DATEADD(YEAR, 1, PlannedStartDate)
        END
    WHERE RepetitionTypeID > 1 AND PlannedStartDate < CAST(GETDATE() AS DATE);
END
